\section{Update Equations}
\writtenby{Sulakna Weerasinghe}

Let $\mathcal{A}_{\mathrm{f}}(s)$ be the feasible joint actions in state $s$ and
$m(s) \in \{0,1\}^{125}$ its mask. Masked scores replace infeasible entries by the lowest
finite value, so $\pi_\theta(a|s) = 0$ for $a \notin \mathcal{A}_{\mathrm{f}}(s)$.

\subsection{DQN}
For a minibatch $\mathcal{B}$ of transitions $(s, a, r, s', d)$ with terminal flag $d$,
\begin{align}
y &= r + \gamma\,(1 - d) \max_{a' \in \mathcal{A}_{\mathrm{f}}(s')} Q_{\bar\theta}(s', a'), \\
\mathcal{L}(\theta) &= \frac{1}{|\mathcal{B}|} \sum_{\mathcal{B}}
\mathrm{Huber}\bigl(Q_\theta(s, a) - y\bigr),
\end{align}
with $\bar\theta \leftarrow \theta$ every 100 environment steps and the $\epsilon$-greedy
action drawn uniformly from $\mathcal{A}_{\mathrm{f}}(s)$.

\subsection{REINFORCE with a learned baseline}
For one episode of length $T$ with reward-to-go
$G_t = \sum_{k=t}^{T-1} \gamma^{k-t} r_k$,
\begin{equation}
\mathcal{L}(\theta, \phi) = -\frac{1}{H}\sum_{t=0}^{T-1} \gamma^t \log \pi_\theta(a_t|s_t)
\bigl(G_t - V_\phi(s_t)\bigr) + \frac{c_v}{T} \sum_{t=0}^{T-1} \bigl(G_t - V_\phi(s_t)\bigr)^2,
\end{equation}
where the baseline enters the actor term as a constant and $c_v = 0.5$. Subtracting
$V_\phi(s_t)$ leaves the gradient unbiased because
$\mathbb{E}_{a \sim \pi}[\nabla_\theta \log \pi_\theta(a|s)\,b(s)] = b(s) \nabla_\theta
\sum_a \pi_\theta(a|s) = 0$ \cite{williams1992, sutton2018}.

\subsection{A2C}
For a segment of $n = 32$ steps starting at $t_0$,
\begin{align}
G^{(n)}_t &= \sum_{k=t}^{t_0+n-1} \gamma^{k-t} r_k + \gamma^{t_0+n-t} (1 - d)\,
V_\phi(s_{t_0+n}), \\
\mathcal{L} &= -\frac{1}{n}\sum_t \gamma^{t} \log \pi_\theta(a_t|s_t)\, A_t
+ c_v \,\overline{(G^{(n)}_t - V_\phi(s_t))^2} - c_e \,\overline{\mathcal{H}[\pi_\theta(\cdot|s_t)]},
\end{align}
with $A_t = G^{(n)}_t - V_\phi(s_t)$ treated as a constant in the actor term, $c_v = 0.5$,
$c_e = 0.05$, and bars denoting means over the segment \cite{mnih2016}.

\subsection{Fire-spread probability}
Each burning neighbour $j$ of an unburnt cell $i$ ignites it independently with probability
$p_{ji} = \min(1, q\,\omega_{ji}\,\tau_i)$. The cell therefore stays unburnt only if every
neighbour fails, with probability $\prod_j (1 - p_{ji})$, which gives (\ref{eq:spread}).
